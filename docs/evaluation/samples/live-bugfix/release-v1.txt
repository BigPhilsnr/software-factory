+/**
+ * Independent regression test: a short link must continue to redirect beyond
+ * 30 visits within a single rolling window. The creation-rate limiter must
+ * not be applied to the redirect path.
+ */
+class ShortenerControllerRedirectThrottlingTest {
+
+    @Test
+    void redirectContinuesPastThirtyVisitsInSameWindow() {
+        ShortenerService service = mock(ShortenerService.class);
+        Link link = new Link(1L, "zz99zz99", "https://example.org", Instant.now());
+        when(service.find("zz99zz99")).thenReturn(Optional.of(link));
+
+        AnalyticsRecorder recorder = mock(AnalyticsRecorder.class);
+        CreationRateLimiter limiter = new CreationRateLimiter(
+            Clock.fixed(Instant.parse("2026-10-01T09:15:00Z"), ZoneOffset.UTC), 30);
+        ShortenerController controller =
+            new ShortenerController(service, limiter, recorder, "http://localhost:8080");
+
+        for (int visit = 1; visit <= 35; visit++) {
+            ResponseEntity<Void> response = controller.redirect("zz99zz99");
+            assertEquals(HttpStatus.FOUND, response.getStatusCode(),
+                "visit " + visit + " must redirect; redirects must not be throttled by the creation-rate limiter");
+        }
+    }
